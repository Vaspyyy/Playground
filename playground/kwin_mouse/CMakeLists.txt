cmake_minimum_required(VERSION 3.20)
project(playground_mouse VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(QT_MAJOR_VERSION 6)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
find_package(KWin REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS DBus)
add_library(playground_mouse MODULE main.cpp)
set_target_properties(playground_mouse PROPERTIES PREFIX "")
target_link_libraries(playground_mouse PRIVATE KWin::kwin Qt6::DBus)
install(TARGETS playground_mouse DESTINATION ${KDE_INSTALL_PLUGINDIR}/kwin/effects/plugins)
